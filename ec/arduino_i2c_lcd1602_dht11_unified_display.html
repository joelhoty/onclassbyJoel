🔌 一、實體硬體接線與螢幕條紋雜訊診斷

在最新的現場上機照片中，Arduino Uno 同時連接了 LCD 1602 與麵包板上的 DHT11：

📷 圖 1-1：Arduino Uno 同時連接 LCD 1602 與 DHT11 實體接線狀況（螢幕顯現上電未初始化條紋）
🔍 為什麼照片中的液晶螢幕會出現垂直條紋雜訊？
  • 根本原因：前一階段燒錄純序列埠版的 DHT_Unified_Sensor.ino 時，程式中完全沒有包含 <Wire.h> 與 LiquidCrystal_I2C 驅動指令。因此 LCD 模組通電後，內部的 HD44780 晶片未收到初始化複位訊號，DDRAM 殘留隨機浮動位元，進而呈現出豎條紋。
  • 解決方案：只要燒錄整合了 Wire.h 與 LiquidCrystal_I2C.h 的新程式，透過 lcd.init(); lcd.init(); lcd.backlight(); lcd.clear(); 執行軟硬體雙重重置，即可徹底消除條紋，正常顯示文字！

💻 四、完整整合程式碼 (`sketch_oct20b.ino` 優化版)

#include <Wire.h> 
#include <LiquidCrystal_I2C.h>
#include <Adafruit_Sensor.h>
#include <DHT.h>
#include <DHT_U.h>

// --- 1. 設定 DHT 感測器 ---
#define DHTPIN 7       // DHT 接在數位腳位 7
#define DHTTYPE DHT11  // 使用 DHT11 感測器
DHT_Unified dht(DHTPIN, DHTTYPE);

// --- 2. 設定 LCD 螢幕 ---
// 預設 I2C 位址為 0x27，16 個字元寬度、2 行顯示
LiquidCrystal_I2C lcd(0x27, 16, 2);

uint32_t delayMS;

void setup() {
  Serial.begin(9600);
  
  // 初始化 LCD 螢幕
  lcd.init();
  lcd.init();          // 雙重初始化，有效清除上電殘留的條紋雜訊
  lcd.backlight();     // 點亮液晶背光
  lcd.clear();         // 清空全螢幕

  // 初始化 DHT 感測器
  dht.begin();
  
  // 取得感測器規格並設定安全延遲時間
  sensor_t sensor;
  dht.temperature().getSensor(&sensor);
  dht.humidity().getSensor(&sensor);
  delayMS = sensor.min_delay / 1000;
  
  // DHT11 讀取間隔建議至少 1.5 秒 ~ 2 秒 (1500ms~2000ms)
  if (delayMS < 1500) {
    delayMS = 1500;
  }
}

void loop() {
  // 感測器讀取間隔延遲
  delay(delayMS);

  sensors_event_t event;

  // ==========================================
  // --- 讀取並顯示溫度 (第一行 Row 0) ---
  // ==========================================
  dht.temperature().getEvent(&event);
  lcd.setCursor(0, 0); // 游標移至第 0 欄、第 0 列 (第一行開頭)
  
  if (isnan(event.temperature)) {
    Serial.println(F("Error reading temperature!"));
    lcd.print(F("TEMP: Error     "));
  } else {
    // 輸出至序列埠監視器
    Serial.print(F("Temperature: "));
    Serial.print(event.temperature);
    Serial.println(F(" *C"));

    // 輸出至 LCD 液晶螢幕第一行
    lcd.print(F("TEMP: "));
    lcd.print(event.temperature, 1); // 顯示到小數點後 1 位 (例如 27.5)
    lcd.write(223);                  // HD44780 內建 ASCII 223 圓圈「°」度數符號
    lcd.print(F("C   "));             // 多留空白確保清除前一次殘留文字，免用 clear()
  }

  // ==========================================
  // --- 讀取並顯示濕度 (第二行 Row 1) ---
  // ==========================================
  dht.humidity().getEvent(&event);
  lcd.setCursor(0, 1); // 游標移至第 0 欄、第 1 列 (第二行開頭)
  
  if (isnan(event.relative_humidity)) {
    Serial.println(F("Error reading humidity!"));
    lcd.print(F("HUMD: Error     "));
  } else {
    // 輸出至序列埠監視器
    Serial.print(F("Humidity: "));
    Serial.print(event.relative_humidity);
    Serial.println(F(" %"));

    // 輸出至 LCD 液晶螢幕第二行
    lcd.print(F("HUMD: "));
    lcd.print(event.relative_humidity, 1); // 顯示到小數點後 1 位 (例如 41.0)
    lcd.print(F(" %   "));                 // 多留空白確保定長覆寫
  }
}
